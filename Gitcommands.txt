chmod +x *.sh
--------------------------------------------
Initialize Local Git Repository and check status
git init

----------------------------------------
Add and commit files
git add .
git commit -m "Initial commit started"
-------------------------------------------
create a repo in github
connect local repo to github
git remote set-url origin https://github.com/dhanainweb97-dev/gitdemoproject.git
--------------------------------
Push the code to github by renaming

git branch -M main
git push -u origin main
------------------------------------
create a branch
git checkout -b feature-login

modify a file, add and commit
echo "Login feature added" >> app.sh
git status

cat app.sh
